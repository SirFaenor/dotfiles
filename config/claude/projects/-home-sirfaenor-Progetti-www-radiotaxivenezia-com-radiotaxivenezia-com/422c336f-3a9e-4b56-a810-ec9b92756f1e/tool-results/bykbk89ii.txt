function forcedownload($file) {
	
	if(file_exists($file) && is_readable($file)) {
		$filename = basename($file);
		if(strpos(strtoupper($_SERVER['HTTP_USER_AGENT']), 'MSIE') !== false && strpos($filename, '.') !== false) {
			$parsename = explode('.', $filename);
			$last = count($parsename) - 1;
			$filename = implode('%2E', array_slice($parsename, 0, $last));
			$filename .= '.'.$parsename[$last];
		}
		header('Content-Type: application/octet-stream');
		header('Content-Disposition: attachment; filename="'.$filename.'"');
		header('Content-Length:'.filesize($file));
		header('Content-Transfer-Encoding: binary');
		if(@$file = fopen($file, "rb")) {
			while(!feof($file))
				echo fread($file, 8192);
			fclose($file);
		}
	}
}

function genera_pass($lunghezza){
	$caratteri = "0123456789abcdefghijklmnopqrstuvxzyw";
	
	$pass_casuale =  '';
	for($i = 1; $i <= $lunghezza; $i++){
		$num = rand(1, 36);
		
		$pass_casuale .= substr($caratteri, ($num-1), 1);
	}
	
	return $pass_casuale;
}

function genera_pass2($length = 8, $mode = 'Ln', $pattern = ''){

	$letters = $numbers = '';	// divido in lettere e numeri per creare dei patterns

	if(!strlen($mode)) $mode = 'Ln';

	$arMode = str_split($mode);

	foreach($arMode as $M):
		switch($M):
			case 'l':
				$letters .= 'abcdefghijklmnopqrstuvxzyw';
				break;
			case 'L':
				$letters .= 'abcdefghijklmnopqrstuvxzywABCDEFGHIJKLMNOPQRSTUVWXYZ';

			case 'n':

				$numbers .= '0123456789';

				break;

				$letters .= '|-_.:!()/\?*';
				break;
			default:
				if(!strlen($letters)) $letters .= 'abcdefghijklmnopqrstuvxzyw';

	endforeach;

	$all = $letters.$numbers;

	if(!strlen($pattern)):						# se non ho passato alcun pattern
		$to_search = $letters.$numbers;			// la stringa in cui cercare sono tutti i caratteri

	$i = 0;

function fixstr($stringa, $tipo = 0){
	
	//Rimpiazzo i target blank per la validazione
	$stringa = str_replace('target', 'data-target', $stringa);
	$stringa = str_replace('_blank', 'blank', $stringa);

	////////////////
	// Tolgo il backslash, altrimenti HTMLPurifier elimina tutto l'attributo
	$stringa = str_replace('\"', '"', $stringa);
	////////////////
	
	$config->set('HTML.AllowedElements', 'table,tr,td,tbody,thead,th,em,strong,br,p,a,ul,li,ol,span,h1,h2,h3,h4,h5,h6,sup,sub,iframe');
	$config->set('HTML.AllowedAttributes', '*.style,*.class,*.id,*.target,a.href,*.width,*.height,*.src,iframe.frameborder,*.wmode');

	$config->set('URI.SafeIframeRegexp', '%^(http:)?//(www.youtube(?:-nocookie)?.com/embed/|player.vimeo.com/video/)%'); //allow YouTube and Vimeo
	
	switch($tipo){
		case 0 : $config->set('AutoFormat.AutoParagraph', false);break;
		case 1 : $config->set('AutoFormat.AutoParagraph', true);break;
	}
	
	$purifier = new HTMLPurifier($config); // Inizializzo
	$stringa = $purifier->purify($stringa); // Purifico
	
	////////////////
	// Ripristino il backslash
	//$stringa = str_replace('"', '\"', $stringa);
	////////////////

	$stringa = htmlentities($stringa,  ENT_QUOTES, 'utf-8');

	if(!get_magic_quotes_gpc())
	{$stringa = addslashes($stringa);}
	$stringa = addcslashes($stringa, "=");

	$stringa = str_replace('&gt;', '>', $stringa);

function google_geocoding($indirizzo){

	// create curl resource
	$ch = curl_init();
	
	// URL
	curl_setopt($ch, CURLOPT_URL, "http://maps.googleapis.com/maps/api/geocode/json?address=".$indirizzo."&sensor=false");
	
	//return the transfer as a string
	curl_setopt($ch, CURLOPT_RETURNTRANSFER, 1);
	
	// Esecuzione azioni
	$output_pagina = curl_exec($ch);
	
	// Chiudo le risorse
	curl_close($ch);
	
	$output_pagina = json_decode($output_pagina);
	
	// Ritorno un array
	if($output_pagina->status == 'OK'){ // Località trovata
		$ritorno = array();
		$ritorno['lat'] = $output_pagina->results[0]->geometry->location->lat;
		$ritorno['lng'] = $output_pagina->results[0]->geometry->location->lng;
		
		return $ritorno;
	}else{ // Nessuna località trovata con esattezza
		$ritorno['lat'] = '';
		$ritorno['lng'] = '';

		return $ritorno;
	}

}

function max_parole($stringa, $max, $strip){
	$stringa_x = explode(' ', $stringa);
	$stringa2 = '';
	
	if(count($stringa_x) > $max){
		for($i = 0; $i < $max; $i++){
			$stringa2 .= $stringa_x[$i].' ';
		}
		$stringa = $stringa2;
		$stringa .= '...'; // Aggiugo i (...) se la stringa  pi lunga del massimo
	}
	if($strip){
		$stringa2 = strip_tags($stringa);
		$stringa = $stringa2;
	} // Tolgo i tags html se indicato
	
	return $stringa;

}

	if(strlen($valore) > 13){$min = substr($valore, 14, 2);}
	if(strlen($valore) > 15){$sec = substr($valore, 17, 2);}
	
	$data = $a.'-'.$m.'-'.$g;
	if(strlen($valore) > 10){
		$data .= ' '.$ora;
		$data .= ':'.$min;

	}

	return $data;

function format_url ($stringa) {
	$stringa = fix_nome_file($stringa);
	
	$stringa = preg_replace("/([^a-zA-Z0-9])/", '-', $stringa);
	$stringa = preg_replace("/([-]{2,})/", '-', $stringa);
	$stringa = preg_replace("/([-]$)/", '-', $stringa);
	
	return $stringa;

function ext_file($filename) {
      // get base name of the filename provided by user
      $filename = basename($filename);

      // break file into parts seperated by .
      $filename = explode('.', $filename);

      // take the last part of the file to get the file extension
      $filename = $filename[count($filename)-1];   

      // find mime type
      return $filename;

function rinomina($percorso, $nome_vecchio){
	if($nome_vecchio){
		$nf = 0;// per uscire dal ciclio in caso di ok
		$i = 0;// variabile ad incremento
		$x = "";// per costruire il nome con la variabile ad incremento
		$nome_file = "";
		
		$ar = pathinfo($nome_vecchio);
		$estensione = $ar['extension'];
		$nome = $ar['filename'];
		
		do{
			if(!file_exists($percorso . fix_nome_file($nome) . $x .'.'. $estensione )){
				$nome_file = fix_nome_file($nome) . $x .'.'. $estensione;
				$nf = 1;
			}
			
			$x = "_$i";
			$i++;
		}while($nf == 0);
		
		return  $nome_file;
	}
}

function carica_file($upload_dir_file, $nome_file, $nome_file_tmp, $peso_file_caricato, $cancellazione, $nome_num = false){

	global $panel_info_;

	if($peso_file_caricato >= $panel_info_['max_weight_upload'] || $peso_file_caricato = 0){
		## GESTIONE ERRORI ##
		$err['upload']['msg_utente'] = "Il file ".$nome_file."  troppo pesante, massimo consentito ".$panel_info_['max_weight_upload']."Bytes";
		return $err;
		exit;
	}
	
	// Rinomina file (con lo stesso nome || con nome numerico)
	if(!$nome_num){
		$nome_file_finale = rinomina($upload_dir_file, $nome_file);
	}else{
		$nome_file_finale = rinomina_con_nome($upload_dir_file, $nome_file, $nome_num);
	}
	
	//definisco la path e nome file
	$path_e_file_finale = $upload_dir_file.$nome_file_finale;
	if(is_uploaded_file($nome_file_tmp)) {
		if(!move_uploaded_file($nome_file_tmp, $path_e_file_finale)){
			## GESTIONE ERRORI ##
			$err['upload']['msg_utente'] = $panel_info_['msg_error_user'];
			$err['upload']['msg_admin'] = "[FUNCTION carica_file]<br />Impossibile spostare il file ".$path_e_file_finale;
			return $err;
			exit;
		}
	}else{
	exit($nome_file_tmp);
		## GESTIONE ERRORI ##
		$err['upload']['msg_utente'] = $panel_info_['msg_error_user'];
		$err['upload']['msg_admin'] = "[FUNCTION carica_file]<br />Problemi nell'upload del file ".$path_e_file_finale;
		return $err;
		exit;
	}
	
	// Cancello un eventuale file
	if($cancellazione){cancella_file($upload_dir_file.$cancellazione);}
	
	//Passo il nuovo nome del file
	return $nome_file_finale;
}

	// === Calcolo della memoria necessaria per ridimensionare ======================
	function setmemoryimage($filename) {
		
		$memoryLimit = 0;
		//$imageInfo = getimagesize(stripslashes($filename)) or dietype($filename);
		$imageInfo = getimagesize($filename);

		$MB = Pow(1024,2);  // number of bytes in 1M
		$K64 = Pow(2,16);    // number of bytes in 64K
		$TWEAKFACTOR = 1.8;  // Or whatever works for you
		$imgChannel = (!empty($imageInfo['channels'])) ? ($imageInfo['channels'] / 8) : (3 / 8);
		$memoryNeeded = round( ( $imageInfo[0] * $imageInfo[1]
											   * $imageInfo['bits']
											   * $imgChannel
								 + $K64
							   ) * $TWEAKFACTOR
							 );
		$memoryHave = memory_get_usage();
		$memoryLimitMB = (integer) ini_get('memory_limit');
		$memoryLimit = $memoryLimit * $MB;
		
		if ( function_exists('memory_get_usage')
			 && $memoryHave + $memoryNeeded > $memoryLimit
		   ) {
		   $newLimit = $memoryLimitMB + ceil( ( $memoryHave
											 + $memoryNeeded
											 - $memoryLimit
											 ) / $MB
										   );
		   ini_set( 'memory_limit', $newLimit . 'M' );
		}
		
	}
	// === Calcolo della memoria necessaria per ridimensionare ======================


	// === Impostazione protezione ridimensionamento senza ritaglio ======================
	//Risetta a 0 UNA tra larghezza e altezza per il ridimensionamento dell'immagine
	//confrontando il rapporto desiderato con quello attuale dell'immmagine,
	//per farla stare sempre nello spazio desiderato senza che la funzione di ridimensionamento effettui il ritaglio
	function reset_dim ($img, $L_max, $A_max) {
		$dim = getimagesize($img);
		$rapp_max_desiderato = $L_max / $A_max;
		$rapp_img = $dim[0] / $dim[1];
		$new_dim = array('L' => $L_max, 'A' => $A_max);
		
		if ( $rapp_img >= $rapp_max_desiderato ) {
			//Se il rapporto delle dimensioni attuali è maggiore o uguale del desiderato, lascio impostata solo la larghezza e "libero" l'altezza
			$new_dim['A'] = 0;
		}
		if ( $rapp_img < $rapp_max_desiderato ) {
			//Se il rapporto delle dimensioni attuali è minore del desiderato, lascio impostata solo l'altezza e "libero" la larghezza
			$new_dim['L'] = 0;
		}
		return $new_dim;
	}

function ridim($src, $dim_lar, $dim_alt, $sl, $sa, $dim_limit = false){

function ridim_thumb($src, $dim_lar, $dim_alt, $xStart, $yStart, $selectedW, $selectedH){
	
	global $panel_info_;
	
	// Recupero le info della immagine
	$immagine = getimagesize($src);
	
	// Se la memoria non basta, assegno la memoria necessaria + 1mb (per la pagina)
	setmemoryimage($src);
	
	
	// Definisco il tipo immagine
	switch($immagine[2]){
		case 2	: $orig = @imagecreatefromjpeg($src);	break;
		case 1	: $orig = @imagecreatefromgif($src); 	break;
		case 3	: $orig = @imagecreatefrompng($src); 	break;
		default	: $orig = @imagecreatefromjpeg($src);
	}
	
	// Setto la compressione del jpg
	$compressione = $panel_info_['jpg_quality_crop'];
	
	$def = imagecreatetruecolor($dim_lar, $dim_alt); 
	imagecopyresampled($def, $orig, 0, 0, $xStart, $yStart, $dim_lar, $dim_alt, $selectedW, $selectedH); 
	
	if(!imagejpeg($def, $src, $compressione)){
		$err['upload']['msg_utente'] = $panel_info_['msg_error_user'];
		$err['upload']['msg_admin'] = "[FUNCTION ridim]<br />Si &egrave; verificato un problema nel ritaglio della immagine.<br />";
	}
	imagedestroy($orig); 
	imagedestroy($def);
		
}
